// 标准初始化，并使用 try-catch 防止报错阻断程序 let supabase; try { // 直接创建客户端实例 supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); } catch (err) { console.error("Supabase 初始化失败:", err); // 如果初始化失败，设置一个空对象防止后续代码报错 supabase = {}; } let myMessages = []; // 存储当前访客发送过的消息ID let pollTimer = null; // 发送消息 async function sendMessage() { // 增加健壮性判断，防止 supabase 未定义时调用 if (!supabase.from) { alert('系统配置错误，请检查 Supabase 初始化代码！'); return; } const input = document.getElementById('userInput'); const nameInput = document.getElementById('userName'); const text = input.value.trim(); if (!text) return; const userName = nameInput.value.trim() || '匿名访客'; // 显示在聊天界面 addMessageToChat('user', text); input.value = ''; document.getElementById('sendBtn').disabled = true; showTypingIndicator(); try { // 插入数据库 const { data, error } = await supabase .from('messages') .insert([{ user_name: userName, content: text }]) .select(); if (error) { console.error('发送失败:', error); alert('发送失败，请重试'); removeTypingIndicator(); document.getElementById('sendBtn').disabled = false; return; } // 记录这条消息的ID，用于后续轮询回复 myMessages.push(data[0].id); removeTypingIndicator(); document.getElementById('sendBtn').disabled = false; // 开始轮询回复 if (!pollTimer) { pollTimer = setInterval(pollReplies, 3000); } } catch (err) { console.log(err); } } // 轮询回复 async function pollReplies() { if (!supabase.from || myMessages.length === 0) return; try { const { data, error } = await supabase .from('messages') .select('id, reply') .in('id', myMessages) .not('reply', 'is', null); if (error || !data) return; data.forEach(msg => { if (msg.reply) { addMessageToChat('assistant', msg.reply); // 从待轮询列表移除 myMessages = myMessages.filter(id => id !== msg.id); } }); // 如果所有消息都有回复了，停止轮询 if (myMessages.length === 0 && pollTimer) { clearInterval(pollTimer); pollTimer = null; } } catch (err) { console.log(err); } } // UI 辅助函数 function addMessageToChat(role, text) { const container = document.getElementById('chatContainer'); const div = document.createElement('div'); div.className = `message ${role}`; div.innerHTML = `
${escapeHtml(text)}
`; container.appendChild(div); container.scrollTop = container.scrollHeight; } function showTypingIndicator() { const container = document.getElementById('chatContainer'); const div = document.createElement('div'); div.className = 'message assistant'; div.id = 'typingIndicator'; div.innerHTML = '
🤖 正在输入...
'; container.appendChild(div); container.scrollTop = container.scrollHeight; } function removeTypingIndicator() { const indicator = document.getElementById('typingIndicator'); if (indicator) indicator.remove(); } function escapeHtml(text) { const div = document.createElement('div'); div.textContent = text; return div.innerHTML; } // 回车发送 document.getElementById('userInput').addEventListener('keypress', (e) => { if (e.key === 'Enter') sendMessage(); }); // 页面加载时，如果有历史消息ID存在localStorage，恢复轮询 window.addEventListener('load', () => { const saved = localStorage.getItem('myMessages'); if (saved) { myMessages = JSON.parse(saved); if (myMessages.length > 0) { pollTimer = setInterval(pollReplies, 3000); } } }); // 保存消息ID到localStorage setInterval(() => { localStorage.setItem('myMessages', JSON.stringify(myMessages)); }, 5000);
